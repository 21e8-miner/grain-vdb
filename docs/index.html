Developer SDKs

Integrate in Swift or Python

First-class packages for Swift Package Manager, Python, and LangChain.

Swift Package (macOS & iOS) SPM 5.9+ 
import GrainVDB

// 1. Initialize embedded vector DB
let db = try GrainVDB(dimension: 128, mode: .exact)

// 2. Add embeddings
try db.addVectors(embeddingMatrix)

// 3. Sub-millisecond retrieval with metadata closure filter
let results = try db.search(query: queryVector, k: 5) { id, meta in
    meta?["category"] as? String == "Privacy"
}

for doc in results {
    print("Match: \(doc.id), Score: \(doc.score)")
}
Python & LangChain Drop-in pip install 
from grainvdb.integrations import GrainVDBVectorStore
from langchain_community.embeddings import OllamaEmbeddings

# Initialize local embedding model & GrainVDB
embeddings = OllamaEmbeddings(model="nomic-embed-text")
vectorstore = GrainVDBVectorStore(embedding=embeddings, dim=768)

# Ingest and query in 0.1ms
vectorstore.add_texts([
    "Apple M2 Ultra features 800 GB/s unified memory.",
    "GrainVDB runs zero-copy vector search on Mac."
])

docs = vectorstore.similarity_search("memory bandwidth", k=1)
print(docs[0].page_content)
